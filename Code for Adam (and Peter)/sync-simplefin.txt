import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

// Helper to identify retirement/investment accounts from SimpleFIN
function isRetirementAccount(acct: any): boolean {
  const name = `${acct.name || ''} ${acct.org?.name || ''}`.toLowerCase();
  return name.includes('acorns') || name.includes('later') || name.includes('ira') || name.includes('roth');
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const body = await req.json();
    const { action } = body;

    // 1. CLAIM ONE-TIME SETUP TOKEN
    if (action === 'claim') {
      const setupToken = String(body.setup_token || '').trim();
      if (!setupToken) {
        return new Response(JSON.stringify({ error: 'Missing setup_token' }), {
          status: 400,
          headers: { ...corsHeaders, 'Content-Type': 'application/json' },
        });
      }

      const claimUrl = atob(setupToken).trim();
      const claimRes = await fetch(claimUrl, {
        method: 'POST',
        headers: { 'Content-Length': '0' },
      });

      if (!claimRes.ok) {
        const txt = await claimRes.text();
        return new Response(
          JSON.stringify({ error: `SimpleFIN claim failed (${claimRes.status}): ${txt}` }),
          { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
        );
      }

      const accessUrl = (await claimRes.text()).trim();
      return new Response(JSON.stringify({ access_url: accessUrl }), {
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      });
    }

    // 2. SYNC TRANSACTIONS & RETIREMENT BALANCES INTO SUPABASE LEDGER
    if (action === 'sync') {
      const ledgerKey = String(body.ledger_key || '').trim();
      const rawAccessUrl = String(body.access_url || '').trim();
      const days = Math.max(1, Math.min(60, Number(body.days) || 14));

      if (!ledgerKey || !rawAccessUrl) {
        return new Response(
          JSON.stringify({ error: 'Missing ledger_key or access_url' }),
          { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
        );
      }

      const parsed = new URL(rawAccessUrl);
      const authHeader = 'Basic ' + btoa(`${decodeURIComponent(parsed.username)}:${decodeURIComponent(parsed.password)}`);
      parsed.username = '';
      parsed.password = '';
      const baseUrl = parsed.toString().replace(/\/$/, '');

      const startDate = Math.floor(Date.now() / 1000) - days * 86400;
      const accountsUrl = `${baseUrl}/accounts?start-date=${startDate}`;

      const sfRes = await fetch(accountsUrl, {
        headers: { Authorization: authHeader },
      });

      if (!sfRes.ok) {
        const errText = await sfRes.text();
        return new Response(
          JSON.stringify({ error: `SimpleFIN /accounts failed (${sfRes.status}): ${errText}` }),
          { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
        );
      }

      const sfData = await sfRes.json();
      if (sfData.errors && sfData.errors.length > 0) {
        return new Response(
          JSON.stringify({ error: `SimpleFIN Bridge message: ${sfData.errors.join('; ')}` }),
          { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
        );
      }

      const supabase = createClient(
        Deno.env.get('SUPABASE_URL')!,
        Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
      );

      const { data: ledger, error: getErr } = await supabase.rpc('get_ledger', { p_key: ledgerKey });
      if (getErr || !ledger) {
        return new Response(
          JSON.stringify({ error: `Could not load ledger: ${getErr?.message || 'not found'}` }),
          { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
        );
      }

      // Initialize retirement object on the ledger if it doesn't exist yet
      if (!ledger.retirement) {
        ledger.retirement = {
          accounts: {},
          contributions: [],
          initialCostBasis: 0, // Editable in UI so "how much it's up" includes pre-SimpleFIN deposits
        };
      }
      if (!ledger.retirement.accounts) ledger.retirement.accounts = {};
      if (!ledger.retirement.contributions) ledger.retirement.contributions = [];

      const existingExternalIds = new Set(
        (ledger.transactions || []).map((t: any) => t.externalId).filter(Boolean)
      );
      const existingContribIds = new Set(
        ledger.retirement.contributions.map((c: any) => c.externalId).filter(Boolean)
      );

      let added = 0;
      let retirementUpdated = false;

      for (const acct of sfData.accounts || []) {
        const isRetirement = isRetirementAccount(acct);

        if (isRetirement) {
          // 1. Save live market balance & metadata
          const currentBalance = parseFloat(acct.balance ?? acct['available-balance'] ?? '0');
          ledger.retirement.accounts[acct.id] = {
            id: acct.id,
            name: acct.name || 'Acorns Later',
            institution: acct.org?.name || 'Acorns',
            balance: isNaN(currentBalance) ? 0 : currentBalance,
            updatedAt: new Date().toISOString(),
          };
          retirementUpdated = true;

          // 2. Capture positive inflows (contributions) into the retirement account
          for (const tx of acct.transactions || []) {
            const extId = `sf_ret_${acct.id}_${tx.id}`;
            if (existingContribIds.has(extId)) continue;

            const amt = parseFloat(tx.amount);
            const desc = (tx.description || tx.payee || '').toLowerCase();

            // Skip dividend reinvestments so they don't count toward IRS contribution limits
            if (isNaN(amt) || amt <= 0 || desc.includes('dividend')) continue;

            const postedTs = Number(tx.posted || tx.transacted_at || Math.floor(Date.now() / 1000));
            const dateStr = new Date(postedTs * 1000).toISOString().slice(0, 10);

            ledger.retirement.contributions.unshift({
              id: 'rc_' + Math.random().toString(36).slice(2, 9),
              externalId: extId,
              accountId: acct.id,
              date: dateStr,
              description: tx.description || tx.payee || 'Roth IRA Contribution',
              amount: amt,
            });
            existingContribIds.add(extId);
            retirementUpdated = true;
          }
          continue; // Keep Acorns internal feed out of regular spending transactions
        }

        // Standard checking/credit card outflow sync
        for (const tx of acct.transactions || []) {
          const extId = `sf_${acct.id}_${tx.id}`;
          if (existingExternalIds.has(extId)) continue;

          const amt = parseFloat(tx.amount);
          if (isNaN(amt) || amt >= 0) continue;

          const postedTs = Number(tx.posted || tx.transacted_at || Math.floor(Date.now() / 1000));
          const dateStr = new Date(postedTs * 1000).toISOString().slice(0, 10);

          ledger.transactions.unshift({
            id: 't_' + Math.random().toString(36).slice(2, 9),
            externalId: extId,
            date: dateStr,
            merchant: tx.description || tx.payee || 'Bank Purchase',
            amount: Math.abs(amt),
            binId: null,
          });
          existingExternalIds.add(extId);
          added++;
        }
      }

      if (added > 0 || retirementUpdated) {
        const { error: saveErr } = await supabase.rpc('save_ledger', {
          p_key: ledgerKey,
          p_payload: ledger,
        });
        if (saveErr) {
          return new Response(
            JSON.stringify({ error: `Failed saving to ledger: ${saveErr.message}` }),
            { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
          );
        }
      }

      return new Response(JSON.stringify({ added, retirementUpdated }), {
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      });
    }

    return new Response(JSON.stringify({ error: 'Unknown action' }), {
      status: 400,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    });
  } catch (err: any) {
    return new Response(JSON.stringify({ error: err.message || String(err) }), {
      status: 400,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    });
  }
});
