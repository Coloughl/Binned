import { serve } from "https://deno.land/std@0.168.0/http/server.ts";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    const RESEND_API_KEY = Deno.env.get("RESEND_API_KEY");
    if (!RESEND_API_KEY) {
      throw new Error("RESEND_API_KEY secret is not configured in Supabase.");
    }

    const {
      to_email,
      target_username,
      auditor_username,
      bin_name,
      merchant,
      amount,
      tx_date,
      auditor_note,
    } = await req.json();

    if (!to_email) {
      return new Response(JSON.stringify({ skipped: true }), {
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    const formattedAmount = `$${Number(amount).toFixed(2)}`;
    const htmlBody = `
      <div style="font-family: sans-serif; max-width: 520px; margin: 0 auto; border: 3px solid #1b1a17; border-radius: 10px; padding: 20px; background: #fffdf7; color: #1b1a17;">
        <div style="display: inline-block; background: #fbe0df; color: #b52b27; border: 2px solid #b52b27; font-weight: 800; padding: 3px 10px; border-radius: 5px; text-transform: uppercase; letter-spacing: 0.06em;">
          Official Bin Audit
        </div>
        <h2 style="margin: 12px 0 8px;">Hey ${target_username}, ${auditor_username} just audited you!</h2>
        <p style="margin: 0 0 14px; color: #575247;">
          Your <strong>${bin_name}</strong> bin went over budget, and your Bin Buddy is demanding an explanation for this purchase:
        </p>
        <div style="background: #f7f3e8; border: 2px solid #1b1a17; border-radius: 8px; padding: 12px; margin-bottom: 14px;">
          <div style="font-size: 16px; font-weight: 800;">${merchant} &mdash; ${formattedAmount}</div>
          <div style="font-size: 12px; color: #575247;">Date: ${tx_date} &bull; Bin: ${bin_name}</div>
          ${auditor_note ? `<div style="margin-top: 8px; padding-top: 8px; border-top: 1px dashed #1b1a17;"><strong>${auditor_username} asks:</strong> "${auditor_note}"</div>` : ""}
        </div>
        <p style="font-size: 13px; margin: 0;">
          Open <strong>Binned &rarr; Accounti-bin-ility</strong> to submit your justification!
        </p>
      </div>
    `;

    const res = await fetch("https://api.resend.com/emails", {
      method: "POST",
      headers: {
        Authorization: `Bearer ${RESEND_API_KEY}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        from: "Binned Audits <onboarding@resend.dev>",
        to: [to_email],
        subject: `🚨 Audit Alert: ${auditor_username} questioned your ${formattedAmount} ${bin_name} purchase`,
        html: htmlBody,
      }),
    });

    const data = await res.json();
    if (!res.ok) {
      throw new Error(data.message || "Failed to send email via Resend");
    }

    return new Response(JSON.stringify({ ok: true, id: data.id }), {
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  } catch (err) {
    return new Response(JSON.stringify({ error: err.message }), {
      status: 400,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  }
});
